## Feature 1 — Portable pasted images (save beside doc + relative path)

**Goal:** When `mdpeek-image-beside-doc` is `1` (default ON), pasted/dropped images are saved to `<docDir>/assets/<hash>.<ext>` and inserted as `![](assets/<hash>.<ext>)` — a portable relative link. Untitled docs and the OFF setting keep the current global + absolute-URL behavior (zero breaking change).

**Why default ON:** the current absolute-URL behavior is a latent bug — images silently break on move/share. New users get the good behavior; existing users can flip it back via Settings → Editor.

### Implementation

**`src/main.js:1414` `insertImageFromBlob`** — branch on the setting:
```js
async function insertImageFromBlob(blob) {
  const doc = store.active();
  if (!doc || !doc.editor) return false;
  const ext = (blob.type.split('/')[1] || 'png').replace('jpeg', 'jpg').replace('svg+xml', 'svg');
  const buf = new Uint8Array(await blob.arrayBuffer());
  const hash = await sha256Hex(buf).catch(() => Math.random().toString(36).slice(2));
  const short = String(hash).slice(0, 10);
  const filename = `img-${short}.${ext}`;

  const besideDoc = localStorage.getItem('mdpeek-image-beside-doc') !== '0';
  const docDir = doc && doc.path ? doc.path.replace(/[\\/][^\\/]+$/, '') : null;
  // Relative mode requires a saved doc (need a folder to save beside).
  if (besideDoc && docDir) {
    try {
      const savedPath = await invoke('save_image', { dir: docDir + '\\assets', filename, bytes: Array.from(buf) });
      md = `![](assets/${filename})`;
    } catch (e) {
      // Fall through to global/data-URL fallback below.
      md = await fallbackImageMarkdown(blob, buf, filename);
    }
  } else {
    md = await fallbackImageMarkdown(blob, buf, filename);
  }
  doc.editor.insertAtCursor(md);
  // ... existing markDirty / persistSoon / scheduleAutoSave
  return true;
}
```

## Feature 2 — Backlinks palette picker

**Goal:** Command palette entry "Backlinks" opens a quick-switcher listing files that link to the current doc — both `[[Foo]]` and `[text](Foo.md)` forms. Selecting one opens that file.

### Implementation

**New `src/lib/backlinks.js`** — pure functions, testable:
```js
// Strip extension + dir from an absolute path → 'Foo' for 'C:\...\Foo.md'.
export function docBasename(path) {
  if (!path) return null;
  const slash = Math.max(path.lastIndexOf('/'), path.lastIndexOf('\\'));
  const file = slash >= 0 ? path.slice(slash + 1) : path;
  return file.replace(/\.(md|markdown|mdx)$/i, '');
}

// Two search queries that together catch both wiki-link and standard-link forms.
// Run in parallel via search_in_folder.
export function backlinkQueries(basename) {
  const safe = basename.replace(/"/g, '');
  return [
    `[[${safe}`,           // [[Foo]] / [[Foo|alias]] / [[Foo.md]]
    `${safe}.md`,          // [text](Foo.md) / (<Foo.md>)
  ];
}

// Dedupe hits by path, sort, format as picker items.
export function formatBacklinkItems(hits, activePath) {
  const seen = new Set();
  const items = [];
  for (const hit of hits) {
    if (!hit || !hit.path || hit.path === activePath) continue;
    if (seen.has(hit.path)) continue;
    seen.add(hit.path);
    const slash = Math.max(hit.path.lastIndexOf('/'), hit.path.lastIndexOf('\\'));
    const name = slash >= 0 ? hit.path.slice(slash + 1) : hit.path;
    const preview = (hit.matches && hit.matches[0] && hit.matches[0].text) || '';
    const hint = preview.length > 60 ? preview.slice(0, 57) + '…' : preview;
    items.push({ label: name, hint, keywords: name + ' ' + hit.path, path: hit.path });
  }
  items.sort((a, b) => a.label.localeCompare(b.label));
  return items;
}
```

**`src/main.js`** — import the helpers, create a backlinks picker (mirror the heading picker from v0.38):

```js
import { docBasename, backlinkQueries, formatBacklinkItems } from './lib/backlinks.js';

const backlinksPicker = initQuickSwitcher(
  () => [],   // populated dynamically in open()
  async (item) => {
    try {
      const content = await invoke('read_file', { path: item.path });
      await openPath(item.path, content);
    } catch (e) { toast('Could not open: ' + fmtErr(e)); }

async function openBacklinks() {
  const doc = store.active();
  if (!doc || !doc.path) { toast('Save the doc first to find backlinks.'); return; }
  const root = localStorage.getItem('mdpeek-explorer-root');
  if (!root) { toast('Open a folder to scan for backlinks.'); return; }
  const name = docBasename(doc.path);
  const queries = backlinkQueries(name);
  const [q1, q2] = await Promise.all(queries.map((q) =>
    invoke('search_in_folder', { root, query: q, caseSensitive: true, maxResults: 200 })
      .catch(() => ({ results: [] }))
  ));
  const hits = [...((q1 && q1.results) || []), ...((q2 && q2.results) || [])];
  const items = formatBacklinkItems(hits, doc.path);
  if (items.length === 0) { toast(`No files link to ${name}.`); return; }
  backlinksPicker.setItems(items);   // dynamic injection (see below)
  backlinksPicker.open();
}
```

## Feature 3 — Slideshow speaker notes

**Goal:** Lines starting with `note:` (case-insensitive, optional space) and `<!-- note: ... -->` HTML comments become speaker notes for that slide. They're stripped from the rendered slide and shown in a toggleable bottom panel, cycled with `N`.

### Implementation

**New `src/lib/slides.js`** — pure parser, testable:
```js
// Extract speaker notes from a slide's markdown source. Returns the cleaned
// markdown (notes removed) and the concatenated note text.
// Supports:  `note: text`  and  `<!-- note: text -->`  (case-insensitive).
const NOTE_LINE = /^[ \t]*note:[ \t]?(.*)$/gmi;
const NOTE_COMMENT = /<!--\s*note:[ \t]?(.*?)\s*-->/gsi;

export function extractSpeakerNotes(slideMd) {
  if (!slideMd) return { cleanMd: '', note: '' };
  const notes = [];
  // HTML comments first (multiline, non-greedy). Replace and collect.
  let clean = slideMd.replace(NOTE_COMMENT, (whole, body) => {
    if (body) notes.push(body.trim());
    return '';
  });
  // Then `note:` lines.
  clean = clean.replace(NOTE_LINE, (whole, body) => {
    if (body) notes.push(body.trim());
    return '';
  });
  // Collapse the blank lines left behind.
  clean = clean.replace(/\n{3,}/g, '\n\n').trim();
  return { cleanMd: clean, note: notes.join('\n') };
}
```

```js
async function createAndOpenMissingLink(fullPath, displayName) {
  const starter = `# ${displayName.replace(/\.(md|markdown|mdx)$/i, '')}\n\n`;
  try {
    await invoke('save_file', { path: fullPath, content: starter });
    await openPath(fullPath, starter);
    refreshTree();
    toast(`Created ${displayName}`);
  } catch (e) {
    toast('Could not create: ' + fmtErr(e));
  }
}
```
